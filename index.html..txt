<script>

let customers = [];
let loans = [];

function addCustomer(){

    let name = document.getElementById("customerName").value;
    let mobile = document.getElementById("mobile").value;

    if(name === ""){
        alert("Enter Customer Name");
        return;
    }

    customers.push({
        name:name,
        mobile:mobile
    });

    document.getElementById("customerCount").innerText = customers.length;

    alert("Customer Added Successfully");

    document.getElementById("customerName").value="";
    document.getElementById("mobile").value="";
}

function calculateLoan(){

    let amount = Number(document.getElementById("loanAmount").value);
    let rate = Number(document.getElementById("interestRate").value);
    let months = Number(document.getElementById("months").value);

    let interest = (amount * rate * months)/100;
    let total = amount + interest;

    document.getElementById("interestResult").innerHTML=
        "Interest = ₹" + interest.toFixed(2) +
        " | Total Payable = ₹" + total.toFixed(2);
}

function saveLoan(){

    let customer=document.getElementById("loanCustomer").value;
    let ornament=document.getElementById("ornament").value;
    let weight=document.getElementById("weight").value;

    let amount=Number(document.getElementById("loanAmount").value);
    let rate=Number(document.getElementById("interestRate").value);
    let months=Number(document.getElementById("months").value);

    let interest=(amount*rate*months)/100;
    let total=amount+interest;

    loans.push({
        customer,
        ornament,
        weight,
        amount,
        interest,
        total
    });

    updateTable();
}

function updateTable(){

    let table=document.getElementById("loanTable");
    table.innerHTML="";

    let totalLoans=0;

    loans.forEach(function(loan){

        totalLoans += loan.amount;

        table.innerHTML += `
        <tr>
            <td>${loan.customer}</td>
            <td>${loan.ornament}</td>
            <td>${loan.weight} gm</td>
            <td>₹${loan.amount}</td>
            <td>₹${loan.interest.toFixed(2)}</td>
            <td>₹${loan.total.toFixed(2)}</td>
        </tr>
        `;
    });

    document.getElementById("loanCount").innerText=loans.length;
    document.getElementById("loanTotal").innerText="₹"+totalLoans.toLocaleString();
}
